#include <tunables/global>

# Profil AppArmor de l'add-on : tout ce qui n'est pas listé ici est refusé, même si quelqu'un
# prenait la main sur l'agent. Le Supervisor remplace le nom « velarys » par l'ID de l'app.
# Pas de S6 ni de shell au démarrage : docker-init lance Node (ENTRYPOINT du Dockerfile), qui
# n'exécute rien d'autre. Tout nouvel accès se teste d'abord en ajoutant `complain` aux flags.
profile velarys flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Arrêt par le Supervisor (SIGTERM du moteur Docker), relais des signaux par docker-init
  signal (receive) peer=unconfined,
  signal (send,receive) peer=@{profile_name},

  # Démarrage : docker-init, puis Node et ses bibliothèques
  /dev/init ix,
  /{usr/,}sbin/docker-init ix,
  /usr/local/bin/node ix,
  /{usr/,}lib/** mr,
  /usr/local/lib/** r,

  # Code de l'agent : lecture seule
  /app/ r,
  /app/** r,

  # Options (écrites par le Supervisor) et identité du hub : seul endroit en écriture
  /data/ r,
  /data/** rw,

  # Ce que Node lit du système : CPU, mémoire, limites du conteneur, aléa
  @{PROC}/** r,
  /sys/fs/cgroup/** r,
  /sys/devices/system/cpu/** r,
  /dev/urandom r,
  /dev/null rw,

  # Résolution de noms et certificats, pour le HTTPS sortant vers le central et l'API Supervisor
  /etc/hosts r,
  /etc/resolv.conf r,
  /etc/nsswitch.conf r,
  /etc/passwd r,
  /etc/group r,
  /etc/ssl/** r,
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
}
